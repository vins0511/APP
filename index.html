📜 구글 스프레드시트에 넣을 Apps Script 샘플 코드 보기 (클릭하여 펼치기) 
// 구글 스프레드시트의 [확장 프로그램] > [Apps Script]에 붙여넣으세요.
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var studentSheet = ss.getSheetByName("학생명단") || ss.getSheets()[0];
  var roleSheet = ss.getSheetByName("역할명단") || (ss.getSheets().length > 1 ? ss.getSheets()[1] : studentSheet);
  
  var students = studentSheet.getRange(2, 1, Math.max(1, studentSheet.getLastRow() - 1), 1).getValues().flat().filter(String);
  var roles = [];
  if (roleSheet !== studentSheet) {
    var roleData = roleSheet.getRange(2, 1, Math.max(1, roleSheet.getLastRow() - 1), 3).getValues();
    roles = roleData.filter(function(r) { return r[0]; }).map(function(r) {
      return { icon: r[0] || "🌿", name: r[1] || "", desc: r[2] || "" };
    });
  }
  
  var result = { status: "success", students: students, roles: roles };
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var params = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var resultSheet = ss.getSheetByName("배정결과") || ss.insertSheet("배정결과");
  
  resultSheet.clear();
  resultSheet.appendRow(["번호", "학생 이름", "담당 역할", "역할 설명", "배정 일시"]);
  
  if (params.assignments) {
    params.assignments.forEach(function(item, idx) {
      resultSheet.appendRow([idx + 1, item.student, item.role.name, item.role.desc, new Date()]);
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "저장 완료!" }))
    .setMimeType(ContentService.MimeType.JSON);
}
